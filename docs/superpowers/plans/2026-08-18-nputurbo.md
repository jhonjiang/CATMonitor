```go
// Package straggler provides an exec source that runs the external
// `straggler` slow-node detector, which writes its result to a path given
// via the `path=` argument. Mirrors the npu_smi exec-source pattern:
// singleton, runner seam for tests, graceful Available()==false when the
// binary is missing.
package straggler

import (
	"context"
	"os/exec"
	"strings"
	"sync"
)

// Source runs the straggler detector.
type Source interface {
	Available() bool
	Run(ctx context.Context, cmdTemplate, resultPath string) error
}

type runner = func(ctx context.Context, cmd string) error

func realRun(ctx context.Context, cmd string) error {
	return exec.CommandContext(ctx, "sh", "-c", cmd).Run()
}

type defaultSource struct {
	runner runner
}

var (
	defaultSrc = &defaultSource{runner: realRun}
	mu         sync.Mutex
)

func Default() Source { return defaultSrc }

func SetMock(fn func(ctx context.Context, cmd string) error) {
	mu.Lock()
	defer mu.Unlock()
	defaultSrc.runner = fn
}

func ResetRunner() {
	mu.Lock()
	defer mu.Unlock()
	defaultSrc.runner = realRun
}

func (s *defaultSource) Available() bool {
	_, err := exec.LookPath("straggler")
	return err == nil
}

func (s *defaultSource) Run(ctx context.Context, cmdTemplate, resultPath string) error {
	cmd := strings.ReplaceAll(cmdTemplate, "{path}", resultPath)
	mu.Lock()
	r := s.runner
	mu.Unlock()
	return r(ctx, cmd)
}
```

```go
// Package npu_turbo provides an exec source that runs the external
// `/var/npu_turbo` frequency-setting tool. Mirrors the straggler/npu_smi
// exec-source pattern: singleton, runner seam, graceful Available()==false
// when the binary is missing.
package npu_turbo

import (
	"context"
	"fmt"
	"os/exec"
	"strconv"
	"strings"
	"sync"
)

type Source interface {
	Available() bool
	SetFreq(ctx context.Context, cmdTemplate string, cardID, freqMHz int) error
}

type runner = func(ctx context.Context, cmd string) error

func realRun(ctx context.Context, cmd string) error {
	return exec.CommandContext(ctx, "sh", "-c", cmd).Run()
}

type defaultSource struct {
	runner runner
}

var (
	defaultSrc = &defaultSource{runner: realRun}
	mu         sync.Mutex
)

func Default() Source { return defaultSrc }

func SetMock(fn func(ctx context.Context, cmd string) error) {
	mu.Lock()
	defer mu.Unlock()
	defaultSrc.runner = fn
}

func ResetRunner() {
	mu.Lock()
	defer mu.Unlock()
	defaultSrc.runner = realRun
}

func (s *defaultSource) Available() bool {
	// Default template's first token is the binary path; caller passes the
	// actual template via SetFreq, so Available() is a best-effort hint.
	_, err := exec.LookPath("/var/npu_turbo")
	return err == nil
}

func (s *defaultSource) SetFreq(ctx context.Context, cmdTemplate string, cardID, freqMHz int) error {
	cmd := strings.ReplaceAll(cmdTemplate, "{id}", strconv.Itoa(cardID))
	cmd = strings.ReplaceAll(cmd, "{freq}", strconv.Itoa(freqMHz))
	mu.Lock()
	r := s.runner
	mu.Unlock()
	if err := r(ctx, cmd); err != nil {
		return fmt.Errorf("npu_turbo SetFreq(id=%d freq=%d): %w", cardID, freqMHz, err)
	}
	return nil
}
```

```go
//go:build linux

package nputurbo

import (
	"encoding/json"
	"strings"
)

// SlowCard is one slow-NPU entry extracted from the straggler result.
type SlowCard struct {
	Hostname string
	ID       int
	Score    float64
}

type calScore struct {
	Score float64 `json:"score"`
}
type npuEntry struct {
	ID  int      `json:"id"`
	Cal calScore `json:"cal"`
}
type nodeResultEntry struct {
	Hostname string     `json:"hostname"`
	NPU      []npuEntry `json:"npu"`
}
type profilerDoc struct {
	Profiler struct {
		NodeResult []nodeResultEntry `json:"node_result"`
		// comm_domain_result ignored
	} `json:"profiler"`
}

// ParseSlowCards extracts the slow-card list from straggler's output.
// It first tries to parse the whole buffer as one profiler doc (handles
// pretty-printed and compact single-doc). If that fails, it splits into
// lines and uses the LAST successfully-parsed line (handles jsonl append).
// An empty node_result yields an empty slice with nil error.
// Returns an error only when no valid doc could be parsed at all.
func ParseSlowCards(data []byte) ([]SlowCard, error) {
	if cards, ok := tryParse(data); ok {
		return cards, nil
	}
	var last []SlowCard
	found := false
	for _, line := range strings.Split(string(data), "\n") {
		line = strings.TrimSpace(line)
		if line == "" {
			continue
		}
		if cards, ok := tryParse([]byte(line)); ok {
			last = cards
			found = true
		}
	}
	if !found {
		return nil, errUnparseable
	}
	return last, nil
}

func tryParse(data []byte) ([]SlowCard, bool) {
	var doc profilerDoc
	if err := json.Unmarshal(data, &doc); err != nil {
		return nil, false
	}
	var out []SlowCard
	for _, n := range doc.Profiler.NodeResult {
		for _, c := range n.NPU {
			out = append(out, SlowCard{Hostname: n.Hostname, ID: c.ID, Score: c.Score})
		}
	}
	return out, true
}

var errUnparseable = jsonUnparseableError{}

type jsonUnparseableError struct{}

func (jsonUnparseableError) Error() string { return "nputurbo: no valid profiler document found in straggler output" }
```

```go
//go:build linux

package nputurbo

// ComputeTargetB returns the target frequency (MHz) to boost a card whose
// current frequency is currentA and slow-card score is `score` (1.0 = normal,
// >1.0 = slower). B = round-to-nearest-step(min(currentA*score, M)).
// ok=false when no boost is warranted: score<=1, or the rounded target
// does not exceed currentA (gain wiped by step quantization).
func ComputeTargetB(currentA int, score float64, maxM, step int) (int, bool) {
	if score <= 1.0 {
		return 0, false
	}
	raw := float64(currentA) * score
	b := roundStep(int(raw+0.5), step) // round half up to nearest step
	if b > maxM {
		b = maxM
	}
	if b <= currentA {
		return 0, false
	}
	return b, true
}

// roundStep rounds v to the nearest multiple of step (ties go up).
func roundStep(v, step int) int {
	if step <= 0 {
		return v
	}
	// round half up
	return ((v + step/2) / step) * step
}
```

```go
package nputurbo

import (
	"context"
	"fmt"
	"log/slog"
	"sort"
	"sync"

	"github.com/Computing-Availability-Tools/CATMonitor/internal/source/npu_turbo"

// Actuator pins NPU cards to boosted frequencies and restores them on
// recovery/shutdown. Per-card original (pre-boost) frequency is saved on
// first boost; restore writes it back. Idempotent: a re-boost to the same
// target is a no-op; a changed target re-applies without touching savedOrig.
type Actuator struct {
	src      npu_turbo.Source
	cmd      string
	logger   *slog.Logger

	mu          sync.Mutex
	savedOrig   map[int]int // id -> pre-boost freq (boost-before value)
	lastApplied map[int]int // id -> last applied target
	ok          bool
}

func NewActuator(src npu_turbo.Source, cmdTemplate string, logger *slog.Logger) *Actuator {
	return &Actuator{
		src:         src,
		cmd:         cmdTemplate,
		logger:      logger,
		savedOrig:   map[int]int{},
		lastApplied: map[int]int{},
		ok:          true,
	}
}

func (a *Actuator) Available() bool { return a.src != nil && a.src.Available() }
func (a *Actuator) Ok() bool        { return a.ok }

// Boost pins card `id` to targetB. currentA is the pre-boost frequency
// (read from snapshot); it is saved as the restore target on first boost.
func (a *Actuator) Boost(ctx context.Context, id, currentA, targetB int) error {
	a.mu.Lock()
	if a.lastApplied[id] == targetB {
		a.mu.Unlock()
		return nil // idempotent
	}
	a.mu.Unlock()
	if err := a.src.SetFreq(ctx, a.cmd, id, targetB); err != nil {
		a.mu.Lock()
		a.ok = false
		a.mu.Unlock()
		if a.logger != nil {
			a.logger.Error("nputurbo: SetFreq failed", "id", id, "target", targetB, "error", err)
		}
		return err
	}
	a.mu.Lock()
	if _, had := a.savedOrig[id]; !had {
		a.savedOrig[id] = currentA
	}
	a.lastApplied[id] = targetB
	a.ok = true
	a.mu.Unlock()
	return nil
}

func (a *Actuator) RestoreID(ctx context.Context, id int) error {
	a.mu.Lock()
	orig, ok := a.savedOrig[id]
	a.mu.Unlock()
	if !ok {
		return nil
	}
	if err := a.src.SetFreq(ctx, a.cmd, id, orig); err != nil {
		a.mu.Lock()
		a.ok = false
		a.mu.Unlock()
		return fmt.Errorf("nputurbo: restore id=%d: %w", id, err)
	}
	a.mu.Lock()
	delete(a.savedOrig, id)
	delete(a.lastApplied, id)
	a.ok = true
	a.mu.Unlock()
	return nil
}

func (a *Actuator) RestoreAll(ctx context.Context) error {
	a.mu.Lock()
	ids := make([]int, 0, len(a.savedOrig))
	for id := range a.savedOrig {
		ids = append(ids, id)
	}
	a.mu.Unlock()
	for _, id := range ids {
		if err := a.RestoreID(ctx, id); err != nil {
			return err
		}
	}
	return nil
}

func (a *Actuator) BoostedIDs() []int {
	a.mu.Lock()
	defer a.mu.Unlock()
	ids := make([]int, 0, len(a.savedOrig))
	for id := range a.savedOrig {
		ids = append(ids, id)
	}
	sort.Ints(ids)
	return ids
}

func (a *Actuator) LastApplied(id int) int {
	a.mu.Lock()
	defer a.mu.Unlock()
	return a.lastApplied[id]
}
```

```go
import (
```

```go
type Config struct {
	Interval          time.Duration
	StragglerCmd      string
	ResultPath        string
	StragglerTimeout  time.Duration
	NpuTurboCmd       string
	NpuTurboTimeout   time.Duration
	MaxFreqMhz        int
	StepMhz           int
	DryRun            bool
	RestoreOnShutdown  bool
	SnapshotDir       string
	Logger            *slog.Logger
}

type Storage interface {
	Write(ms []collector.Metric) error
}

type StragglerSource interface {
	Available() bool
	Run(ctx context.Context, cmdTemplate, resultPath string) error
}

type Controller struct {
	cfg      Config
	stragg   StragglerSource
	actuator *Actuator
	store    Storage
	logger   *slog.Logger
}

func NewController(cfg Config, stragg StragglerSource, actuator *Actuator, store Storage) *Controller {
	if cfg.Logger == nil {
		cfg.Logger = slog.Default()
	}
	return &Controller{cfg: cfg, stragg: stragg, actuator: actuator, store: store, logger: cfg.Logger}
}

func (c *Controller) Run(ctx context.Context) {
	c.logger.Info("nputurbo controller started",
		"interval", c.cfg.Interval, "dry_run", c.cfg.DryRun,
		"straggler_cmd", c.cfg.StragglerCmd, "result_path", c.cfg.ResultPath)
	if c.cfg.Interval <= 0 {
		c.cfg.Interval = 60 * time.Second
	}
	t := time.NewTicker(c.cfg.Interval)
	defer t.Stop()
	for {
		select {
		case <-ctx.Done():
			return
		case <-t.C:
			c.tick(time.Now())
		}
	}
}

func (c *Controller) Restore() {
	if !c.cfg.RestoreOnShutdown {
		return
	}
	ctx, cancel := context.WithTimeout(context.Background(), 30*time.Second)
	defer cancel()
	if err := c.actuator.RestoreAll(ctx); err != nil {
		c.logger.Error("nputurbo shutdown restore failed", "error", err)
	}
}

func (c *Controller) tick(now time.Time) {
	if c.cfg.SnapshotDir == "" {
		c.logger.Warn("nputurbo: snapshot dir not set; cannot read aicore_freq; no-op")
		return
	}
	// 1. Run straggler to refresh result file.
	sctx, scancel := context.WithTimeout(context.Background(), c.cfg.StragglerTimeout)
	defer scancel()
	if err := c.stragg.Run(sctx, c.cfg.StragglerCmd, c.cfg.ResultPath); err != nil {
		c.logger.Error("nputurbo: straggler run failed; no-op this cycle", "error", err)
		return
	}
	data, err := os.ReadFile(c.cfg.ResultPath)
	if err != nil {
		c.logger.Error("nputurbo: read result failed; no-op", "path", c.cfg.ResultPath, "error", err)
		return
	}
	cards, perr := ParseSlowCards(data)
	if perr != nil {
		c.logger.Error("nputurbo: parse result failed; no-op", "error", perr)
		return
	}
	// 2. Read current freqs from snapshot_npu.json → map[id]A.
	freqs := readAicoreFreqs(c.cfg.SnapshotDir)
	// 3. Reconcile: restore boosted-but-not-listed; boost listed.
	desired := map[int]bool{}
	for _, sc := range cards {
		desired[sc.ID] = true
	}
	// Restore disappeared first.
	for _, id := range c.actuator.BoostedIDs() {
		if !desired[id] {
			rctx, rcancel := context.WithTimeout(context.Background(), c.cfg.NpuTurboTimeout)
			if err := c.actuator.RestoreID(rctx, id); err != nil {
				c.logger.Error("nputurbo: restore failed", "id", id, "error", err)
			}
			rcancel()
		}
	}
	// Boost / refresh.
	for _, sc := range cards {
		a, ok := freqs[sc.ID]
		if !ok {
			c.logger.Warn("nputurbo: no aicore_freq in snapshot for card; skip", "id", sc.ID)
			continue
		}
		b, okB := ComputeTargetB(a, sc.Score, c.cfg.MaxFreqMhz, c.cfg.StepMhz)
		if !okB {
			continue
		}
		if c.cfg.DryRun {
			c.logger.Info("nputurbo dry-run: would boost", "id", sc.ID, "A", a, "score", sc.Score, "B", b)
			continue
		}
		bctx, bcancel := context.WithTimeout(context.Background(), c.cfg.NpuTurboTimeout)
		if err := c.actuator.Boost(bctx, sc.ID, a, b); err != nil {
			c.logger.Error("nputurbo: boost failed", "id", sc.ID, "error", err)
		}
		bcancel()
	}
	c.emitMetrics(now)
}

// readAicoreFreqs reads snapshot_npu.json and returns map[npu_id]aicore_freq.
func readAicoreFreqs(snapshotDir string) map[int]int {
	out := map[int]int{}
	cs, err := snapshot.ReadComp(filepath.Join(snapshotDir, "snapshot_npu.json"))
	if err != nil {
		return out
	}
	for _, m := range cs.Metrics {
		if m.Component == "npu" && m.Name == "aicore_freq" && m.Labels != nil {
			if id, err := atoi(m.Labels["npu_id"]); err == nil {
				out[id] = int(m.Value)
			}
		}
	}
	return out

func (c *Controller) emitMetrics(now time.Time) {

		return

	for _, id := range c.actuator.BoostedIDs() {
		boostCount++
		_ = id
	}
	ms = append(ms,
		collector.Metric{Component: "nputurbo", Name: "boost_active", Value: boolFloat(boostCount > 0), Unit: "", Timestamp: now},
		collector.Metric{Component: "nputurbo", Name: "boost_count", Value: float64(boostCount), Unit: "", Timestamp: now},
		collector.Metric{Component: "nputurbo", Name: "actuator_ok", Value: boolFloat(c.actuator.Ok()), Unit: "", Timestamp: now},
	)
	if filtered := metrics.Filter(ms); len(filtered) > 0 {
		if err := c.store.Write(filtered); err != nil {
			c.logger.Error("nputurbo: metric write failed", "error", err)
		}
	}
}
```

```go
func RunOnce(cfg Config, stragg StragglerSource, actuator *Actuator) Snapshot {
	cfg.DryRun = true
	c := NewController(cfg, stragg, actuator, nil)
	rows, freqs, _ := c.planBoosts()
	out := Snapshot{DryRun: true, ActuatorOK: actuator.Available()}
	for id, a := range freqs {
		row := BoostRow{ID: id, A: a}
		out.Rows = append(out.Rows, row)

	}

			b, okB := ComputeTargetB(out.Rows[i].A, score, cfg.MaxFreqMhz, cfg.StepMhz)

			out.Rows[i].WouldBoost = okB

	}
	return out
}
```
